\begin{lem}[Expression Inversion]
	\label{lem:inv_expr}
	For all {$ x^i \in \mathcal{LV}$, and $\trace, \trace' \in \tdom$, and an expression $\expr$} if
	$ \forall z^j \in \mathcal{LV} / \{x^i\} \sthat  
	\env(\trace) z = \env(\trace') z $, and if
	\begin{itemize}
		\item $\expr$ is an arithmetic expression $\aexpr$,
		% \\ 
		and $\config{\trace, \aexpr} \aarrow v $ and 
	$\config{\trace', \aexpr} \aarrow v' $ with $v' \neq v$, 
	then $ x $ is in the free variables of $\aexpr$ and $i$ is the latest label for $x$ 
    in $\trace$, i.e., $x \in \kw{FV}(\aexpr)$ and $i = \llabel(\trace) x$.
%
	\item $\expr$ is a boolean expression $\bexpr$,
	% \\
  and $\config{\trace, \bexpr} \barrow v $ and 
 $\config{\trace', \bexpr} \barrow v' $ with $v' \neq v$, then $ x $ is in the free variables of $\bexpr$ and $i$ is the latest label for $x$ 
 in $\trace$, i.e., $x \in \kw{FV}(\bexpr)$ and $i = \llabel(\trace) x$.
% 
	\item $\expr$ is a query expression $\qexpr$,
	% \\
	and $\config{\trace, \qexpr} \qarrow \qval $ and 
	$\config{\trace', \qexpr} \qarrow \qval' $ with $\qval \neq_{q} \qval'$, then $ x $ is in the free variables of $\qexpr$ and $i$ is the latest label for $x$ 
    in $\trace$, i.e., $x \in \kw{FV}(\qexpr)$ and $i = \llabel(\trace) x$.
\end{itemize}	%
	\end{lem}
    Proof Summary:
    \\
    To show $x \in \kw{FV}(\aexpr)$, by showing contradiction ($\forall \trace, \trace'$ in second hypothesis  $v = v'$)
     if $x \notin \kw{FV}(\aexpr)$.
     \\
    To show $i = \llabel(\trace)$, by showing contradiction ($\forall \trace, \trace'$ in second hypothesis  $v = v'$ ) 
    if $j = \llabel(\trace) x$ and $i \neq j$.
    \begin{proof}
		Take two arbitrary traces $\trace, \trace' \in \tdom$, and an expression $\expr$ satisfying
		$ \forall z^j \in \mathcal{LV} / \{x^i\} \sthat  
		\env(\trace) z = \env(\trace') z $, we have the following three cases.
    \caseL{$\expr$ is an arithmetic expression $\aexpr$}
	We have $\config{\trace, \bexpr} \barrow v $ and 
	$\config{\trace', \bexpr} \barrow v' $ with $v' \neq v$ from the lemma hypothesis.
	\\
	To show $x \in \kw{FV}(\qexpr)$ and $i = \llabel(\trace) x$: 
	\\
	Assuming $x \notin \kw{FV}(\aexpr)$,
	% by $\config{\trace, \aexpr} \aarrow v $	and $\config{\trace', \aexpr} \aarrow v'$, 
	since
	%  $x \notin \kw{FV}(\aexpr)$ and 
	$ \forall z^j \in \mathcal{LV} / \{x^i\} \sthat  
		\env(\trace) z = \env(\trace') z $,
	we know $v = v'$, which is contradicted to $v' \neq v$.
	\\
	Then we know $x \in \kw{FV}(\qexpr)$.
	\\
	Assuming $j = \llabel(\trace) x \land i \neq j$,
	by 
	% $\config{\trace, \aexpr} \aarrow v $	and $\config{\trace', \aexpr} \aarrow v'$, 
	% since $x \notin \kw{FV}(\aexpr)$ and 
	$ \forall z^j \in \mathcal{LV} / \{x^i\} \sthat  
		\env(\trace) z = \env(\trace') z $, we know 
		$\env(\trace) x = \env(\trace') x$, i.e., 
	\\
	$\forall z^j \in \mathcal{LV} \sthat  \env(\trace) z = \env(\trace') z$.
	\\
	Then by the determination of the evaluation, 
	% and 
	% $\config{\trace, \aexpr} \aarrow v $ and $\config{\trace', \aexpr} \aarrow v'$, 
	we know $v = v'$, which is contradicted to $v' \neq v$.
	\\
	Then we know $i = \llabel(\trace) x$.

    \caseL{$\expr$ is a boolean expression $\bexpr$}
	This case is proved trivially in the same way as the case of the arithmetic expression.
	\caseL{$\expr$ is a query expression $\qexpr$}
	This case is proved trivially in the same way as the case of the arithmetic expression.
\end{proof}
\begin{lem}[Expression Inversion Generalization]
	\label{lem:inv_expr_gnl}
	For all subset of the labelled variables $\mathcal{DF} \subset \mathcal{LV}$, and $x^i \in (\mathcal{LV} \setminus \mathcal{DF})$,
	and an expression $\expr$, if 
	\begin{itemize}
		\item $\expr$ is an arithmetic expression $\aexpr$,
		% \\ 
		and for all $z^j \in \mathcal{LV} \setminus \mathcal{DF}, \trace, \trace' \in \ftdom, v, v'$ such that 
		$\env(\trace) z = \env(\trace') z$, and 
		$
		\config{\trace, \aexpr} \aarrow v$, and $\config{\trace', \aexpr} \aarrow v'$ with $v = v'$;
		and for all $z^j \in \mathcal{LV} / (\mathcal{DF} \cup \{x^i\} )$ 
		there exist $\trace, \trace' \in \ftdom, v, v'$ such that 
		$\env(\trace) z = \env(\trace') z$, and 
		$
		\config{\trace, \aexpr} \aarrow v$, and $\config{\trace', \aexpr} \aarrow v'$ with $v \neq v'$,
		then $x \in \kw{FV}(\aexpr)$ and $i = \llabel(\trace) x$.
		\[
			\begin{array}{l}
			\forall \mathcal{DF} \subset \mathcal{LV},  x^i \in (\mathcal{LV} \setminus \mathcal{DF}), \aexpr \sthat 
			\\ \quad
			\forall z^j \in \mathcal{LV} \setminus \mathcal{DF}, \trace, \trace' \in \ftdom, v, v' \sthat  
			\\ \quad
			\env(\trace) z = \env(\trace') z \land 
			\config{\trace, \aexpr} \aarrow v \land \config{\trace', \aexpr} \aarrow v' \land v = v'
			\\ \quad
			\implies 
			\forall z^j \in \mathcal{LV} / (\mathcal{DF} \cup \{x^i\} ) \sthat  
			\exists \trace, \trace' \in \ftdom, v, v'\sthat  
			\\ \quad
			\env(\trace) z = \env(\trace') z \land 
			\config{\trace, \aexpr} \aarrow v \land \config{\trace', \aexpr} \aarrow v' \land v \neq v'
			\\ \qquad
			\implies x \in \kw{FV}(\aexpr) \land i = \llabel(\trace) x
			\end{array}
		\]
	\item $\expr$ is a boolean expression $\bexpr$,
	and for all $ z^j \in \mathcal{LV} \setminus \mathcal{DF}, \trace, \trace' \in \ftdom, v, v'$ such that 
	$ \env(\trace) z = \env(\trace') z \land 
	\config{\trace, \bexpr} \barrow v \land \config{\trace', \bexpr} \barrow v' \land v = v'$;
	and for all
	$ z^j \in \mathcal{LV} / (\mathcal{DF} \cup \{x^i\} ) \sthat  
	 \exists \trace, \trace' \in \ftdom, v, v'\sthat  
	\env(\trace) z = \env(\trace') z \land 
	\config{\trace, \bexpr} \barrow v \land \config{\trace', \bexpr} \barrow v' \land v \neq v'$
	then 
	 $x \in \kw{FV}(\bexpr) \land i = \llabel(\trace) x$
	% \\
	\[
		\begin{array}{l}
		\forall \mathcal{DF} \subset \mathcal{LV},  x^i \in (\mathcal{LV} \setminus \mathcal{DF}), \bexpr \sthat 
		\\ \quad
		\forall z^j \in \mathcal{LV} \setminus \mathcal{DF}, \trace, \trace' \in \ftdom, v, v' \sthat  
		\\ \quad
		\env(\trace) z = \env(\trace') z \land 
		\config{\trace, \bexpr} \barrow v \land \config{\trace', \bexpr} \barrow v' \land v = v'
		\\ \quad
		\implies 
		\forall z^j \in \mathcal{LV} / (\mathcal{DF} \cup \{x^i\} ) \sthat  
		 \exists \trace, \trace' \in \ftdom, v, v'\sthat  
		 \\ \quad
		 \env(\trace) z = \env(\trace') z \land 
		\config{\trace, \bexpr} \barrow v \land \config{\trace', \bexpr} \barrow v' \land v \neq v'
		\\ \qquad
		\implies x \in \kw{FV}(\bexpr) \land i = \llabel(\trace) x
		\end{array}
	\]
% 
	\item $\expr$ is a query expression $\qexpr$,
	and for all $\mathcal{DF} \subset \mathcal{LV},  x^i \in (\mathcal{LV} \setminus \mathcal{DF}), \qexpr$ such that 
	for all $ z^j \in \mathcal{LV} \setminus \mathcal{DF}, \trace, \trace' \in \ftdom, \qval, \qval' \sthat  
 \env(\trace) z = \env(\trace') z \land 
 \config{\trace, \qexpr} \qarrow \qval \land \config{\trace', \qexpr} \qarrow \qval' \land \qval =_q \qval'$;
 and for all 
	$ z^j \in \mathcal{LV} / (\mathcal{DF} \cup \{x^i\} ) \sthat  
  \exists \trace, \trace' \in \ftdom, \qval, \qval'\sthat  
 \env(\trace) z = \env(\trace') z \land 
 \config{\trace, \qexpr} \qarrow \qval \land \config{\trace', \qexpr} \qarrow \qval' \land \qval \neq_{q} \qval'$,
 then  $x \in \kw{FV}(\qexpr) \land i = \llabel(\trace) x$.
	% \\
	\[
		\begin{array}{l}
		\forall \mathcal{DF} \subset \mathcal{LV},  x^i \in (\mathcal{LV} \setminus \mathcal{DF}), \qexpr \sthat 
		\\ \quad
		\forall z^j \in \mathcal{LV} \setminus \mathcal{DF}, \trace, \trace' \in \ftdom, \qval, \qval' \sthat  
		\\ \quad
		\env(\trace) z = \env(\trace') z \land 
		\config{\trace, \qexpr} \qarrow \qval \land \config{\trace', \qexpr} \qarrow \qval' \land \qval =_q \qval'
		\\ \quad
		\implies 
		\forall z^j \in \mathcal{LV} / (\mathcal{DF} \cup \{x^i\} ) \sthat  
		 \exists \trace, \trace' \in \ftdom, \qval, \qval'\sthat  
		 \\ \quad
		 \env(\trace) z = \env(\trace') z \land 
		\config{\trace, \qexpr} \qarrow \qval \land \config{\trace', \qexpr} \qarrow \qval' \land \qval \neq_{q} \qval'
		\\ \qquad
		\implies x \in \kw{FV}(\qexpr) \land i = \llabel(\trace) x
		\end{array}
	\]
	\end{itemize}
	\end{lem}
	%
Proof Summary: 
\\
To show $x \in \kw{FV}(\aexpr)$, by showing contradiction ($\forall \trace, \trace'$ in second hypothesis  $v = v'$)
 if $x \notin \kw{FV}(\aexpr)$.
 \\
To show $i = \llabel(\trace)$, by showing contradiction ($\forall \trace, \trace'$ in second hypothesis  $v = v'$ ) 
if $j = \llabel(\trace) x$ and $i \neq j$.
\begin{proof}
	Taking an arbitrary expression $\expr$,
	 we have the following three cases.
\caseL{$\expr$ is an arithmetic expression $\aexpr$}
Taking an arbitrary set of labelled variables 
	$\mathcal{DF} \subset \mathcal{LV}$, $x^i \in (\mathcal{LV} \setminus \mathcal{DF})$ satisfies:
	\\
	$\forall z^j \in \mathcal{LV} \setminus \mathcal{DF}, \trace, \trace' \in \ftdom, v, v' \sthat  
	\env(\trace) z = \env(\trace') z \land 
	\config{\trace, \aexpr} \aarrow v \land \config{\trace', \aexpr} \aarrow v' \land v = v' ~ (1)
	$
	\\
	and 
	$\forall z^j \in \mathcal{LV} \setminus (\mathcal{DF} \cup \{x^i\} ) \sthat  
	\exists \trace, \trace' \in \ftdom, v, v'\sthat  
	\env(\trace) z = \env(\trace') z \land 
	\config{\trace, \aexpr} \aarrow v \land \config{\trace', \aexpr} \aarrow v' \land v \neq v' ~ (2) 
	$,
	\\
	Let $\trace, \trace' \in \ftdom, v, v'$ be the two traces and values satisfies hypothesis $(2)$.
	\\
	To show: $x \in \kw{FV}(\aexpr) \land i = \llabel(\trace) x$:
	\\
Assuming $x \notin \kw{FV}(\aexpr)$, we know from the Inversion Lemma~\ref{lem:inv_expr} of the arithmetic expression case,
\\
$\forall z^j \in \mathcal{LV} \setminus \{x^i\}, \trace, \trace' \in \ftdom, v, v' \sthat  
\env(\trace) z = \env(\trace') z \land 
\config{\trace, \aexpr} \aarrow v \land \config{\trace', \aexpr} \aarrow v' \land v = v'$.
\\
Then with the hypothesis $(1)$, we know:
\\
$\forall z^j \in \mathcal{LV} \setminus (\mathcal{DF} \cup \{x^i\} ), \trace, \trace' \in \ftdom, v, v'\sthat  
\env(\trace) z = \env(\trace') z \land 
\config{\trace, \aexpr} \aarrow v \land \config{\trace', \aexpr} \aarrow v' \land v = v'$
\\
This is contradicted to the hypothesis $(2)$.
\\
Then we know $x \in \kw{FV}(\expr)$.
\\
Assuming $j = \llabel(\trace) x \land i \neq j$,
by hypothesis $(2)$ where 
% $\config{\trace, \aexpr} \aarrow v $	and $\config{\trace', \aexpr} \aarrow v'$, 
% since $x \notin \kw{FV}(\aexpr)$ and 
$ \forall z^j \in \mathcal{LV} \setminus (\mathcal{DF} \cup \{x^i\} )  \sthat \env(\trace) z = \env(\trace') z $, 
we know $\env(\trace) x = \env(\trace') x$, i.e., 
\\
$\forall z^j \in  \mathcal{LV} \setminus (\mathcal{DF}  ) \sthat  \env(\trace) z = \env(\trace') z$.
\\
Then we have $v' = v$ by hypothesis $(1)$, which is contradicted to $v' \neq v$.
\\
Then we know $i = \llabel(\trace) x $.

\caseL{$\expr$ is a boolean expression $\bexpr$}
This case is proved trivially in the same way as the case of the arithmetic expression.
\caseL{$\expr$ is a query expression $\qexpr$}
This case is proved trivially in the same way as the case of the arithmetic expression.
\end{proof}
\begin{lem}[Event Inversion]
\label{lem:inv_event}
For all $c\in \cdom, \trace_0 \in \ftdom, \event \in \eventset$such that 
$\config{c, \trace_0} \rightarrow^* \config{ \clabel{\eskip}^{l'}, \trace_0 \tracecat \trace_1}$, 
and $\event \eventin \trace_1$, if 
\begin{itemize}
	\item $\event \in \eventset^{\asn}$, then either
	\begin{itemize}
	 \item there exists $\trace_1' \in \ftdom, c' \in \cdom, \expr$ such that
\[
\begin{array}{l}
		\config{c, \trace_0} \rightarrow^* \config{ [\assign{x}{\expr}]^l;c', \trace_0  \tracecat  \trace'} 
		\rightarrow^\rname{assn}
		\config{c', \trace_0 \tracecat \trace_1'\tracecat [\event]} \rightarrow^{*}
		\config{ \clabel{\eskip}^{l'}, \trace_0  \tracecat  \trace_1}
\end{array}
\]
\item or there exists $\trace_1' \in \ftdom, c' \in \cdom, \qexpr$ such that 
\[
\begin{array}{l}
		\config{c, \trace_0} \rightarrow^* \config{ [\assign{x}{\query(\qexpr)}]^l;c', \trace_0 \tracecat \trace_1'} 
		\\ \qquad
		\rightarrow^\rname{query}
		\config{c', \trace_0  \tracecat  \trace_1' \tracecat [\event] } \rightarrow^{*}
		\config{ \eskip, \trace_0 \tracecat \trace_1}
	% \big)
\end{array}
\]
\end{itemize}

\item $\event\in \eventset^{\test}$ then either 
\begin{itemize}
\item there exists $\trace_1' \in \ftdom, c', c_t, c_f, c'' \in \cdom, \bexpr$ such that
\[
\begin{array}{l}
	% \big( 
	% 	\exists \trace_1' \in \ftdom, c', c_t, c_f, c'' \in \cdom, \bexpr \sthat 
		\config{c, \trace_0} \rightarrow^* \config{\eif ([b]^l, c_t, c_f);c', \trace_0 \tracecat \trace_1'} 
		\\ \qquad
		\rightarrow^\rname{if-b}
		\config{c'', \trace_0 \tracecat \trace_1'\tracecat [\event] } \rightarrow^{*}
		\config{ \clabel{\eskip}^{l'}, \trace_0 \tracecat \trace_1} 
\end{array}
\]
\item or there exists $ \trace_1' \in \ftdom, c', c_w, c'' \in \cdom, \bexpr$ such that 
\[
	\begin{array}{l}
		\config{c, \trace_0} \rightarrow^* \config{ \ewhile \clabel{b}^l \edo (c_w);c', \trace_0 \tracecat  \trace_1'} 
		\\ \qquad
		\rightarrow^\rname{while-b}
		\config{c'', \trace_0 \tracecat \trace_1'\tracecat [\event] } \rightarrow^{*}
		\config{ \clabel{\eskip}^{l'}, \trace_0  \tracecat \trace_1}
% 	\big)
\end{array}
\]
\end{itemize}
\end{itemize}
%
\end{lem}
Proof Summary: trivially by induction on $c$ and enumerate all operational semantic rules.

		\\
		This case is proved.

\begin{lem}[Reachable Varibale Inversion]
\label{lem:inv_live}
For a well-formed program $c \in \cdom$, an initial trace $\trace_0 \in \ftdom_0(c)$, an execution trace $\trace \in \tdom $, and a labeled variable $x^l \in \lvar(c)$,
 if 
$\config{c, \trace_0} \xrightarrow{}^* \config{c', \trace}$
and
$\llabel(\trace) x = l $, then $x^l \in \live(\absinit(c), c)$.
%
\[
	\begin{array}{l}
	\forall c \in \cdom , \trace_0 \in \ftdom_0(c), \trace \in \ftdom,x^l \in \lvar(c) \sthat 
	\config{c, \trace_0} \xrightarrow{}^* \config{c', \trace}
	\land
	% \forall x^l \in \lvar(c) \sthat  \llabel(\trace) x = l \implies (x^l \in \live^{\entry_{c'}}(c))
	% \forall x^l \in \lvar(c) \sthat  
	\llabel(\trace) x = l 
	\\ \qquad \qquad \implies x^l \in \live(\absinit(c), c)
	\end{array}
\]
\end{lem}
Proof Summary: 
If a variable with the label which is the latest one in the trace,
Then by the environment definition, the value associated to this labelled variable is read from the trace.
\\
Then this labelled variable must be reachable at the point of $\entry_{c'}$, i.e., 
% $x^l \in \live^{\entry_{c'}}(c)$.
$x^l \in \live(\absinit(c), c)$.
\begin{proof}
	Take arbitrary $c \in \cdom $, an initial trace $\trace_0 \in \ftdom_0(c)$, an execution trace $\trace \in \tdom $ satisfying 
	$\config{c, \trace_0} \xrightarrow{}^* \config{c', \trace}$, 
	and an arbitrary $x^l \in \lvar(c)$ satisfying $\llabel(\trace) x = l$.
	\\
	By definition of $\llabel$, we know $\trace$ has the form $\trace_{a} \tracecat [(x, l, v)] \tracecat \trace_{b}'$
	for some $\trace_{a} , \trace_{b}' \in \tdom$ and $v$.
	\\
	And the variable $x$ doesn't show up in all the events in $\trace_b'$.
%
\\
	Then, by the environment definition, we know:
	$\env(\trace) x = v$, i.e., $x^l$ is 
	reachable at the point of 
	% $\entry_{c'}$.
	$\absinit(c)$.
	\\
	By the $in(l)$ operator define in Section~\ref{sec:alg_edgegen}, we know $x^l$ is in the $in(\absinit(c)$ for prpgram $c$.
	\\
	% By the $\live$ definition, 
	Since $\live(\absinit(c), c)$ is a stabilized closure of $in(l)$ for $c$,
	we know 
	% $x^l \in \live^{\entry_{c'}}(c)$.
	$x^l \in \live(\absinit(c), c)$.
	\\
	This lemma is proved.
\end{proof}
