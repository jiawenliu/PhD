    %
    \[
      \begin{array}{l}
      \forall \trace_0 \in \ftdom_0(c), \trace \in \tdom, c, c' \in \cdom \sthat
      % \Big( 
        \config{c, \trace_0} \rightarrow \config{c', \trace} 
      % \lor  \config{c, \trace_0} \uparrow^{\infty} \trace \Big)

\subsection{Proof of Lemma~\ref{lem:lvar_unique}}
\label{apdx:lvar_unique}
\begin{lemma}[Uniqueness of the Labeled Variables]
  Given any program $c \in \cdom$ with its label variable set $\lvar(c)$,
  every two labeled variables in this set are different.
  \[
    \forall c \in \cdom, x^i, y^j \in \lvar(c) \sthat i \neq j.
    \]
\end{lemma}
\begin{proof}
  This is proved directly by the consistency property of the command label.
\end{proof}


\subsection{Proof of Corollary~\ref{coro:query_equal}}
     \label{apdx:query_equal}
\begin{corollary}
      % \label{coro:query_equal}
  For any event $\event \in \eventset$  that is generated from a query request, 
  there exist another event $\event' \in \eventset$ 
  % and traces $\trace_1, \trace_2 \in \ftdom$
  such that 
  $\event$ and $\event'$ are equivalent but different in their query results.
  \[
    \forall \event \in \eventset \sthat 
    \pi_4(\event) \neq \bullet
    \implies \exists 
    \event' \in \eventset \sthat 
    \event = \event' \land  \pi_3(\event) \neq \pi_3(\event')
  \]
\end{corollary}   
\begin{proof}
  Let $\event = (x, l, v, \qval) \in \eventset$ be arbitrary event that  is generated from a query request for some variable $x$, label $l$ and a query result $v$.
  \\
  By the query evaluation rule \rname{query}, we know $v \in \mathbb{N}$.
  \\
  Let $v' = v + 1$, we construct another event $\event' = (x, l, v', \qval) \in \eventset$.
  \\
  By definition~\ref{def:query_equal}, we know $\event =_{q}\event'$.
  \\
  We also know $ \pi_3(\event) \neq \pi_3(\event')$ since $v' = v + 1 \neq v$.
  \\
  So we have this corollary proved.
\end{proof}  


\subsection{Proof of Corollary~\ref{coro:aqintrace}}
\label{apdx:aqintrace}
\begin{corollary}
  % \label{coro:aqintrace}
  For any event $\event \in \eventset$  that is generated from a query request and a trace $\trace \in \ftdom$ contains this event
  such that $\trace$ can be expressed
  in the form of $\trace_1 \tracecat [\event] \tracecat \trace_2 = \trace $ for some traces $\trace_1, \trace_2 \in \ftdom$,
  then there exist another event $\event' \in \eventset$ such that  $ \pi_3(\event) \neq \pi_3(\event') $
  and the same $\trace$ has a different form
  $\trace_1 \tracecat [\event'] \tracecat \trace_2 $.
  \[
    \begin{array}{l}
    \forall \event \in \eventset, \trace, \trace_1, \trace_2 \in \ftdom \sthat 
    \pi_4(\event) \neq \bullet \land \trace_1 \tracecat [\event] \tracecat \trace_2 = \trace
    \\ \qquad 
    \implies \exists 
    \event' \in \eventset \sthat 
    \trace_1 \tracecat [\event'] \tracecat \trace_2 = \trace  \land \pi_3(\event) \neq \pi_3(\event') 
    \end{array}
  \]
\end{corollary}

\begin{proof}
    Let $\event = (x, l, v, \qval) \in \eventset$ be arbitrary event that  is generated from a query request command for some variable $x$, label $l$, query result $v$ and query value $\qval$.
    \\
    Let $ \trace \in \ftdom$ be the trace that contains $\event$ and two finite traces $\trace_1, \trace_2$ satisfying
    \[
      \trace = \trace_1 \tracecat [\event] \tracecat \trace_2.
      \]
    Then it is sufficient to show
    \[
      \exists \event' \in \eventset \sthat 
      \trace_1 \tracecat [\event'] \tracecat \trace_2 = \trace  \land \pi_3(\event) \neq \pi_3(\event') 
    \]
    %
    By Corollary~\ref{coro:query_equal}, let $\event'$ be the event such that 
    \[
      \event =_{q}\event' \land  \pi_3(\event) \neq \pi_3(\event').
    \]
    Then by definition~\ref{def:event_belong}, we know 
    $\event' \in \trace$ and 
    \[
      \trace_1 \tracecat [\event'] \tracecat \trace_2 = \trace
    \]
    Then we have this corollary proved.
\end{proof}
