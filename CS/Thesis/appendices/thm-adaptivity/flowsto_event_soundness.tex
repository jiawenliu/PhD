We first introduce some conventional operators as follows, which will be used in the proof.
\begin{defn}[Subtrace]
  Subtrace: $[ : ] : \mathcal{{T} \to \eventset \to \eventset \to \ftdom}$ 
  \[
    \trace[\event_x : \event_y] \triangleq
    \left\{
    \begin{array}{ll} 
    \trace'[\event_x: \event_y]             & \trace = \event :: \trace' \land \event \eventneq \event_x \\
    \event_x :: \trace'[:\event_y]  & \trace = \event :: \trace' \land \event \eventeq \event_x \\
    {[]} & \trace = [] \\
    \end{array}
    \right.
  \]
  For any trace $\trace$ and two events $\event_x, \event_y \in \eventset$,
  $\trace[\event_x : \event_y]$ takes the subtrace of $\trace$ starting with $\event_x$ and ending with $\event_y$ including $\event_x$ and $\event_y$.
  \\
  We use $\trace[:\event_y] $ as the shorthand of subtrace starting from head and ending with $\event_y$, and similary for $\trace[\event_x:]$.
  \[
    \trace[:\event] \triangleq
    \left\{
    \begin{array}{ll} 
   \event' :: \trace'[: \event]             & \trace = \event' :: \trace' \land \event' \eventneq \event \\
    \event'  & \trace = \event' :: \trace' \land \event' \eventeq \event \\
    {[]}  & \trace = [] 
    \end{array}
    \right.
  % \]
  % \[
    \quad
    \trace[\event: ] \triangleq
    \left\{
    \begin{array}{ll} 
    \trace'[\event: ]     & \trace =  \event' :: \trace' \land \event \eventneq \event' \\
    \event' :: \trace'  & \trace = \event' :: \trace' \land \event \eventeq \event' \\
    {[ ] } & \trace = []
    \end{array}
    \right.
  \]
\end{defn}

The \emph{Latest Label}
operator $\llabel : \ftdom \to \mathcal{V} \to \mathbb{N} \cup \{ \bot\}$,
takes a trace and a variable as input and returns the label of the latest assignment event which assigns value to that variable. 
  Its behavior is defined as follows,
  \begin{defn}[Latest Label]
    \label{def:latest_label}
    Given the trace $\trace \in \tdom$ and a variable $x \in \mathcal{VAR}$,
    the \emph{latest label} of $x$ is computed as follows.
    \[
      \begin{array}{l}
    \llabel(\trace  :: (x, l, \_, \_)) x \triangleq l
    \qquad~
    \llabel(\trace  :: (y, l, \_, \_)) x \triangleq \llabel(\trace ) x, y \neq x
    % &
    \\
    \llabel(\trace :: (b, l, v, \bullet)) x \triangleq \llabel(\trace) x
    \qquad~
    \llabel({[]}) x \triangleq \bot
    \end{array}
    \]
  \end{defn}

  \begin{thm}[$\eventdep$ implies $\flowsto$]
    \label{thm:flowsto_event_soundness}
    For every $D \in \dbdom , c \in \cdom, \trace \in \ftdom \sthat  \forall \event_x, \event_y \in \eventset \sthat 
    \event_x, \event_y \in \eventset^{\asn}$, 
    if $\exists \trace' \in \ftdom \sthat  \trace = [\event_x] \tracecat \trace' \tracecat [\event_y]$ and 
    $\eventdep(\event_x, \event_y, \trace, c, D) $, then
    there exists $z_1^{r_1}, \ldots, z_n^{r_n} \in \lvar(c)$ with $n \geq 0$ such that   
    $\flowsto(x^i,  z_1^{r_1}, c) 
    \land \ldots \land \flowsto(z_n^{r_n}, y^j, c)$
    %
    \[
      \begin{array}{l}
        \forall D \in \dbdom , c \in \cdom, \trace \in \ftdom, \event_x, \event_y \in \eventset
        % , x^i, y^j \in \lvar(c) 
        \sthat 
        % \\ \qquad 
      %   \pi_1(\event_x)^{\pi_2(\event_x)} = x^i
      %   \land 
      %   \pi_1(\event_y)^{\pi_2(\event_y)} = y^j
      % \land 
        %  \exists \trace' \in \ftdom \sthat  \trace = [\event_x] \tracecat \trace' \tracecat [\event_y]
        % \\ \qquad 
        % \implies
        \eventdep(\event_x, \event_y, \trace, c, D) 
        \\ \quad 
        \implies 
        \exists n \in \mathbb{N}, z_1^{r_1}, \ldots, z_n^{r_n} \in \lvar(c) \sthat 
        \\ \qquad 
         n \geq 0 \land
        \flowsto(\pi_1(\event_x)^{\pi_2(\event_x)},  z_1^{r_1}, c)

           \Big(
          \config{{c}, \vtrace_0} \rightarrow^{*} 
              \config{{c}_1, \vtrace_1 \tracecat [\event_x]}  \rightarrow^{*} 
              \config{c_2,  \trace_1 \tracecat [\event_x;\event_b] \tracecat  \trace_3} 
            \\   
           & \qquad \bigwedge 
            \config{{c}_1, \vtrace_1 \tracecat [\event_x']}  \rightarrow^{*} 
            \config{c_2,  \vtrace_1 \tracecat [\event_x'] \tracecat \trace' \tracecat [(\neg \event_b)] \tracecat \trace_3'} 
            \\
            & \qquad \bigwedge  \tlabel({\trace_3}) \cap \tlabel({\trace_3'}) = \emptyset
             \land \vcounter(\trace') ~  \pi_2(\event_b) = \vcounter(\trace) ~  \pi_2(\event_b)
              \land \event_y \eventin \trace_3
            \land \event_y \not\eventin \trace_3'
           \Big)
         \end{array}
         \]
         Then by Lemma~\ref{lem:ctldep_inv} we know 
         \[
          \forall z \in \kw{FV}(\pi_1(\event_b)) \sthat  
          \exists l \in \mathbb{N} \sthat  
          \flowsto(z^l, \pi_1(\event_y)^{\pi_2(\event_y)},c)~(\star)
        \]
         By Lemma~\ref{lem:inv_indepevents_ctl}, we know
         \[
          \begin{array}{l}
            \big(
              \pi_1(\event_x) \in \kw{FV}(\pi_1(\event_b)) \land {\pi_2(\event_x)} = \llabel(\trace)
            \big)
          \\
          \lor 
          \big(
            \exists \event \in \trace_{ih} \sthat 
            \event \in \eventset^{\asn} \land \pi_1(\event) \in  \kw{FV}(\pi_1(\event_b)) 
          \land \dep(\event_x, \event, \trace_{ih}[\event_x:\event], c)
          \big)
          \end{array}
        \]
          Then we have two subcases.
          \subcaseL{$ \pi_1(\event_x) \in \kw{FV}(\pi_1(\event_y)) \land {\pi_2(\event_x)} = \llabel(\trace)$}
          In this case, by Definition~\ref{def:feasible_flowsto} and $(\star)$ we know
          \[
            \flowsto(\pi_1(\event_x)^{\pi_2(\event_x)}, \pi_1(\event_y)^{\pi_2(\event_y)}, c)
          \]
          \subcaseL{$\exists \event \in \trace_{ih} \sthat \event \in \eventset^{\asn} \land  \pi_1(\event) \in  \kw{FV}(\pi_1(\event_b)) 
          \land \dep(\event_x, \event, \trace_{ih}[\event_x:\event], c)$}
          In this case, let $\event_{ih}\in \trace_{ih} $ be the assignment event such that
          \[
            \pi_1(\event_{ih}) \in  \kw{FV}(\pi_1(\event_b)) 
            \land \dep(\event_x, \event, \trace_{ih}[\event_x:\event_{ih}], c)
          \]
          By induction hypothesis on $\dep(\event_x, \event, \trace_{ih}[\event_x:\event_{ih}], c)$ we know
          \[
            \begin{array}{l}
              \exists n \in \mathbb{N}, z_1^{r_1}, \ldots, z_n^{r_n} \in \lvar(c) 
              \sthat  n \geq 0 
              \\ \quad  \land
              \flowsto(\pi_1(\event_x)^{\pi_2(\event_x)},  z_1^{r_1}, c) 
              \land \ldots \land \flowsto(z_n^{r_n}, \pi_1(\event_{ih})^{\pi_2(\event_{ih})}, c) 
            \end{array}
            \]
            % \\
            Let $m \geq 0 \in \mathbb{N}$ and $z_1^{r_1}, \ldots, z_m^{r_m} \in \lvar(c) $ be 
            $m$ labeled variables such that
            \[
              \flowsto(\pi_1(\event_x)^{\pi_2(\event_x)},  z_1^{r_1}, c) 
            \land \ldots \land \flowsto(z_m^{r_m}, \pi_1(\event_{ih})^{\pi_2(\event_{ih})}, c).
            \]
            By $\pi_1(\event_{ih}) \in  \kw{FV}(\pi_1(\event_b)) $ and $(\star)$ and Definition~\ref{def:feasible_flowsto}, we also know
            \[
              \flowsto(\pi_1(\event_{ih})^{\pi_2(\event_{ih})}, \pi_1(\event_y)^{\pi_2(\event_y)}, c)
              \]
            Then we have the following by the assumption.
            \[
              \begin{array}{l}
                % \exists n \in \mathbb{N}, z_1^{r_1}, \ldots, z_n^{r_n} \in \lvar(c) \sthat  n \geq 1\land
              \flowsto(\pi_1(\event_x)^{\pi_2(\event_x)},  z_1^{r_1}, c) 
              \land \ldots \land \flowsto(z_m^{r_m}, \pi_1(\event_{ih})^{\pi_2(\event_{ih})}, c) )
               \\ \quad  
               \land  \flowsto(\pi_1(\event_{ih})^{\pi_2(\event_{ih})}, \pi_1(\event_y)^{\pi_2(\event_y)}, c) 
              \end{array}
              \]
              Then let $n = m + 1$,
              and $z_{m+1}^{r_{m+1}} = (\event_{ih})^{\pi_2(\event_{ih})}$, we know
              \[
                \begin{array}{l}
                  % \exists n \in \mathbb{N}, z_1^{r_1}, \ldots, z_n^{r_n} \in \lvar(c) \sthat  n \geq 1\land
                \flowsto(\pi_1(\event_x)^{\pi_2(\event_x)},  z_1^{r_1}, c) 
                % \\ \quad  
                \land \ldots 
                % \land \flowsto(z_m^{r_m}, \pi_1(\event_{ih})^{\pi_2(\event_{ih})}, c) )
                %  \\ \quad  
                 \land  \flowsto(z_{m+1}^{r_{m+1}}, \pi_1(\event_y)^{\pi_2(\event_y)}, c) 
                \end{array}
                \]            
                Then we know
                \[
                  \begin{array}{l}
                    \exists n \in \mathbb{N}, z_1^{r_1}, \ldots, z_n^{r_n} \in \lvar(c) \sthat  n \geq 1 
                    \\ \quad  
                    \land
                  \flowsto(\pi_1(\event_x)^{\pi_2(\event_x)},  z_1^{r_1}, c) 
                  \land \ldots 
                  % \land \flowsto(z_m^{r_m}, \pi_1(\event_{ih})^{\pi_2(\event_{ih})}, c) )
                  %  \\ \quad  
                   \land  \flowsto(z_{n}^{r_{n}}, \pi_1(\event_y)^{\pi_2(\event_y)}, c) 
                  \end{array}
                  \]               
              This case is proved.

      \end{case}

\end{proof}
