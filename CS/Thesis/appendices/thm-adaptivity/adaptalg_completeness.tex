\begin{thm}[Conditional Completeness of $\pathsearch$]
  \label{thm:adaptalg_pcomplete}
  For any well-formed program $c$ with its \emph{estimated dependency graph} $\progG(c)$,
  if $\progG(c)$ is acyclic directed, then $\pathsearch(c)$ computes a tight bound 
  on the length of the longest walk on $\progG(c)$.
  \[
    \begin{array}{l}
      \forall c \in \cdom, \sccgraph \subseteq_{\kw{graph}} \progG(c),
    \trace_0 \in \ftdom_0(c), m \in \mathbb{N} \cup \{\infty\} \sthat
    \\ \qquad   
      \config{\pathsearch(c), \trace_0} \earrow m
      \implies
    \max\{\qlen(k)(\trace_0) | k \in \walks(\progG)\} = m
  \end{array}
  \]
\end{thm}
