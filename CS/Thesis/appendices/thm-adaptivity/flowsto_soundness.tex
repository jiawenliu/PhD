\begin{thm}[$\flowsto$ Soundness]
\label{thm:flowsto_soundness}
Given a program ${c}$, for all  $ x^i, y^j \in \lvar(c)$, if $\vardep(x^i, y^j, {c})$,
then
%  either $\flowsto(x^i, y^j, c)$ 
% or 
there exist $z_1^{r_1}, \ldots, z_n^{r_n} \in \lvar(c)$ with $n \geq 0$ such that   
$\flowsto(x^i,  z_1^{r_1}, c) 
\land \ldots \land \flowsto(z_n^{r_n}, y^j, c)$
%
\[
\begin{array}{l}
  \forall x^i, y^j \in \lvar(c) \sthat
  \vardep(x^i, y^j, {c})
  \\ \quad \implies
  \Big( \exists n \in \mathbb{N}, z_1^{r_1}, \ldots, z_n^{r_n} \in \lvar(c) \sthat  n \geq 0 
  \\ \qquad \qquad \land
  \flowsto(x^i,  z_1^{r_1}, c) 
  \land \ldots \land \flowsto(z_n^{r_n}, y^j, c) \Big)
\end{array}
\]
\end{thm}
The proof is summarized informally as below.

For  arbitrary two $x^i, y^j \in \lvar(c)$, with \emph{Variable May-Dependency} relation, in order to show there exists a 
"flows-to chain" relation from the static analysis results from $x^i$ to $y^j$, it is sufficient to show:
  \\
  1. \textbf{One-step Soundness.} $x^i$ is directly used in the expression of the assignment command associated to $y^j$, or a boolean
  expression of the guard for a $\eif$ or $\ewhile$ command with the assignment command associated to $y^j$ showing up in the body of that command, 
  we call it, $x^i$ directly flows to $y^j$,
  i.e.,
  \[\flowsto(x^i, y^j, {c})\]

  2. \textbf{Multiple-step Soundness.}, 
  there exists another labelled variable $z^l$ with \emph{variable May-Dependency} relation on $x^i$ and 
  $z^l$ directly flows to $y^j$, where the \emph{variable May-Dependency} relation between $x^i$ and $z^l$
   implies a "sub flowsto-chain" from $z^i$ to $z^l$, 
  i.e., 
  \[
    \begin{array}{l}
    \exists z^l \in \lvar(c).
  % \Big(
    \vardep(x^i, z^l, {c})   
  % \\ \implies
  %  \exists n \in \mathbb{N}, z_1^{r_1}, \ldots, z_n^{r_n} \in \lvar(c) \sthat  n \geq 0 
  %  \\ \qquad
  %  \land
  % \flowsto(x^i,  z_1^{r_1}, c) 
  % \land \ldots \land \flowsto(z_n^{r_n}, z^l, c)
  % \Big)
  \land  \flowsto(z^l, y^j, {c})
\end{array}
    \]
%
%
By definition of $\vardep(x^i, y^j, {c})$, 
let $D \in \dbdom$ be the dataset,
and $\trace \in \ftdom$, $\event_x, \event_y$ be the trace and two events 
with $\pi_1(\event_x) = x$ and $\pi_1(\event_y) = y$,
such that
\[ \eventdep(\event_x, \event_y, \trace, c, D). 
  \]
Then we show the soundness w.r.t. the \emph{Event May-dependency} relation
through the
\textbf{One-step Soundness} and the \textbf{Multiple-step Soundness} cases.
\[
  \begin{array}{l}
    \eventdep(\event_x, \event_y, \trace, c, D)
    \implies
      \Big( 
      \flowsto(x^i, y^j, c) 
      \Big)
\\ \qquad \lor
\Big( 
  \exists \event \in \trace \sthat  
% \left( 		
  \event \in \eventset^{\asn}
  \land
  \eventdep(\event_x, \event, \trace[\event_x:\event], c, D)
  % \\ \quad  
  \land 
\flowsto(\pi_1(\event)^{\pi_2(\event)}, y^j, c) 
\Big)
\end{array}
  \]
% $$
% \begin{array}{l}
% (\flowsto(\pi_1(\event_1)^{\pi_2(\event_1)}, \pi_1(\event_2)^{\pi_2(\event_2)}, c)
% \\ \qquad 
% \lor 
% (\exists \event_z \in \trace' \sthat  \event_z \in \eventset^{\asn} \land 
%     \eventdep(\event_x, \event_z, \trace[\event_x: \event_z], c, D)
%     \\ \qquad 
%     \implies 
%     \exists n \in \mathbb{N}, z_1^{r_1}, \ldots, z_n^{r_n} \in \lvar(c) \sthat  
%     \\ \qquad
%     n \geq 0 \land
%     \flowsto(x^i,  z_1^{r_1}, c) 
%     \land \ldots \land \flowsto(z_n^{r_n},  \pi_1(\event_z)^{\pi_2(\event_z)}, c)
%   )
%   \\ \qquad \qquad
%   \land 
%     \flowsto(\pi_1(\event_z)^{\pi_2(\event_z)}, \pi_1(\event_2)^{\pi_2(\event_2)}, c) 
% \end{array}
% $$
% %
% Then, it is clearer to prove the following two lemmas,
% \\
% 1. Existence of a middle event:
% in Lemma~\ref{lem:depevents_exist}.
% %
% \\
% 2. The middle event with a sub-trace implies a "sub flowsto-chain", 
%  by induction on the trace $\trace$
% %
% \[
%   \begin{array}{l}
% 		\forall D \in \dbdom , c \in \cdom, \trace \in \ftdom \sthat  \forall \event_1, \event_2 \in \eventset \sthat 
%     \event_1, \event_2 \in \eventset^{\asn} \land 
% 		 \exists \trace' \in \ftdom \sthat  \trace = [\event_1] \tracecat \trace' \tracecat [\event_2]
% 		\\ \qquad 
%     \land
% 		\eventdep(\event_1, \event_2, \trace, c, D) 
% 		\\ \qquad 
% 		\implies 
%     \exists n \in \mathbb{N}, z_1^{r_1}, \ldots, z_n^{r_n} \in \lvar(c) \sthat  
%     \\ \qquad n \geq 0 \land
% 		\flowsto(\pi_1(\event_1)^{\pi_2(\event_1)},  z_1^{r_1}, c) 
%     \land \ldots \land \flowsto(z_n^{r_n}, \pi_1(\event_2)^{\pi_2(\event_2)}, c) 
% 	\end{array}
% \]
% with the induction hypothesis:
% \[
%   \begin{array}{l}
%     \forall \event_{ih1}, \event_{ih2} \in \trace \sthat  \event_{ih1}, \event_{ih2} \in \eventset^{\asn} 
%     \land
% 		 \exists \trace' \in \ftdom \sthat  
%      \trace[\event_{ih1}:\event_{ih2}] = [\event_{ih1}] \tracecat \trace' \tracecat [\event_{ih2}]
% 		\\ \qquad 
%     \land
% 		\eventdep(\event_{ih1}, \event_{ih2}, \trace[\event_{ih1}:\event_{ih2}], c, D) 
% 		\\ \quad 
% 		\implies 
%     \exists n \in \mathbb{N}, z_1^{r_1}, \ldots, z_n^{r_n} \in \lvar(c) \sthat  
%     \\ \qquad 
%     n \geq 0 \land
% 		\flowsto(\pi_1(\event_{ih1})^{\pi_2(\event_{ih1})},  z_1^{r_1}, c) 
%     \land \ldots \land \flowsto(z_n^{r_n}, \pi_1(\event_{ih2})^{\pi_2(\event_{ih2})}, c) 
%   \end{array}

\begin{proof}
  Taking arbitrary $x^i, y^j \in \lvar(c)$,
%
%
by definition of $\vardep(x^i, y^j, {c})$, 
let $D \in \dbdom$ be the dataset,
$\trace \in \ftdom$, $\event_x, \event_y$ be the trace and two events such that,
%  the definition, 
% with 
\[\pi_1(\event_x)^{\pi_2(\event_x)} = x^i
\land 
\pi_1(\event_y)^{\pi_2(\event_y)} = y^j
\land \eventdep(\event_x, \event_y, \trace, c, D). \]  
Then, it is sufficient to show,
$$
\begin{array}{l}
  \eventdep(\event_x, \event_y, \trace, c, D) 
  \\ \quad 
  \implies 
  \exists n \in \mathbb{N}, z_1^{r_1}, \ldots, z_n^{r_n} \in \lvar(c) \sthat  
  \\ \quad 
  n \geq 0 \land
  \flowsto(x^i,  z_1^{r_1}, c) 
  \land \ldots \land \flowsto(z_n^{r_n}, y^j, c) 
\end{array}
$$
%
By Theorem~\ref{thm:flowsto_event_soundness}, we have this theorem proved.
\end{proof}
